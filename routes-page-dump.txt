// app/routes/page.tsx

"use client"

import { useState, useEffect, useCallback } from "react"
import { useRouter } from "next/navigation"
import { useAuth } from "@/lib/auth-context"
import { useSidebar } from "@/lib/sidebar-context"
import { Sidebar } from "@/components/sidebar"
import { Header } from "@/components/header"
import { ActiveRouteCard } from "@/components/routes/active-route-card"
import { AddLoadDialog } from "@/components/routes/add-load-dialog"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Tabs, TabsList, TabsTrigger } from "@/components/ui/tabs"
import {
  Loader2,
  Search,
  RefreshCw,
  Route,
  CheckCircle2,
  Clock,
  Truck,
  Package,
  TrendingUp,
} from "lucide-react"
import { toast } from "sonner"

interface RouteData {
  id: string
  driverName: string
  vehiclePlate: string
  ordersCount: number
  completedOrders: number
  totalDistance: number
  totalWeight: number
  totalPrice: number
  availableCapacity: number
  status: string
  orders: any[]
}

export default function RoutesPage() {
  const { user, isLoading: authLoading } = useAuth()
  const { isCollapsed } = useSidebar()
  const router = useRouter()

  const [routes, setRoutes] = useState<RouteData[]>([])
  const [isLoading, setIsLoading] = useState(true)
  const [searchQuery, setSearchQuery] = useState("")
  const [activeTab, setActiveTab] = useState<"active" | "completed">("active")
  const [addLoadRoute, setAddLoadRoute] = useState<RouteData | null>(null)

  useEffect(() => {
    if (!authLoading && !user) {
      router.push("/")
    }
  }, [user, authLoading, router])

  const fetchRoutes = useCallback(async () => {
    setIsLoading(true)
    try {
      const statusFilter = activeTab === "active"
        ? "confirmed,in_transit,loading,unloading"
        : "delivered"

      const res = await fetch(`/api/orders?status=${statusFilter}`)
      const data = await res.json()

      if (data.success) {
        const routeMap = new Map<string, any>()

        for (const order of data.orders || []) {
          if (!order.routeId) continue

          if (!routeMap.has(order.routeId)) {
            routeMap.set(order.routeId, {
              id: order.routeId,
              driverId: order.assignedDriverId,
              vehicleId: order.assignedVehicleId,
              orders: [],
            })
          }

          routeMap.get(order.routeId).orders.push(order)
        }

        const enrichedRoutes: RouteData[] = []

        for (const [routeId, routeData] of routeMap) {
          const orders = routeData.orders

          let driver = null
          let vehicle = null

          if (routeData.driverId) {
            try {
              const driverRes = await fetch(`/api/drivers/${routeData.driverId}`)
              const driverData = await driverRes.json()
              driver = driverData.driver
            } catch {}
          }

          if (routeData.vehicleId) {
            try {
              const vehicleRes = await fetch(`/api/vehicles/${routeData.vehicleId}`)
              const vehicleData = await vehicleRes.json()
              vehicle = vehicleData.vehicle
            } catch {}
          }

          const totalWeight = orders.reduce((sum: number, o: any) => sum + (o.weight || 0), 0)
          const vehicleCapacity = vehicle?.capacity || 0

          enrichedRoutes.push({
            id: routeId,
            driverName: driver?.name || "РќРµ РЅР°Р·РЅР°С‡РµРЅ",
            vehiclePlate: vehicle?.plate || "вЂ”",
            ordersCount: orders.length,
            completedOrders: orders.filter((o: any) => o.status === "delivered").length,
            totalDistance: orders.reduce((sum: number, o: any) => sum + (o.distance || 0), 0),
            totalWeight,
            totalPrice: orders.reduce((sum: number, o: any) => sum + (o.price || 0), 0),
            availableCapacity: Math.max(0, vehicleCapacity - totalWeight),
            status: orders.some((o: any) =>
              ["in_transit", "loading", "unloading"].includes(o.status)
            )
              ? "in_progress"
              : "pending",
            orders,
          })
        }

        setRoutes(enrichedRoutes)
      }
    } catch (error) {
      console.error("Failed to fetch routes:", error)
      toast.error("РћС€РёР±РєР° Р·Р°РіСЂСѓР·РєРё РјР°СЂС€СЂСѓС‚РѕРІ")
    } finally {
      setIsLoading(false)
    }
  }, [activeTab])

  useEffect(() => {
    if (user) {
      fetchRoutes()
    }
  }, [user, fetchRoutes])

  const filteredRoutes = routes.filter((route) => {
    if (!searchQuery) return true
    const q = searchQuery.toLowerCase()
    return (
      route.driverName.toLowerCase().includes(q) ||
      route.vehiclePlate.toLowerCase().includes(q) ||
      route.orders.some(
        (o: any) =>
          o.routeFrom?.toLowerCase().includes(q) || o.routeTo?.toLowerCase().includes(q)
      )
    )
  })

  const stats = {
    totalRoutes: routes.length,
    inProgress: routes.filter((r) => r.status === "in_progress").length,
    totalRevenue: routes.reduce((sum, r) => sum + r.totalPrice, 0),
    totalCapacity: routes.reduce((sum, r) => sum + r.availableCapacity, 0),
  }

  if (authLoading || !user) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <Loader2 className="animate-spin" />
      </div>
    )
  }

  return (
    <div className="min-h-screen bg-background">
      <Sidebar />
      <div
        className="transition-all duration-300"
        style={{ paddingLeft: isCollapsed ? "80px" : "256px" }}
      >
        <Header />
        <main className="p-6 space-y-6">
          <div className="flex justify-between items-center">
            <div>
              <h1 className="text-2xl font-bold">РђРєС‚РёРІРЅС‹Рµ РјР°СЂС€СЂСѓС‚С‹</h1>
              <p className="text-muted-foreground">РЈРїСЂР°РІР»РµРЅРёРµ СЂРµР№СЃР°РјРё Рё РґРѕРіСЂСѓР·Р°РјРё</p>
            </div>
          </div>

          <div className="grid grid-cols-4 gap-4">
            <div className="bg-card border rounded-xl p-4">
              <div className="flex items-center gap-3">
                <div className="p-2 rounded-lg bg-blue-500/10">
                  <Route className="h-5 w-5 text-blue-500" />
                </div>
                <div>
                  <p className="text-2xl font-bold">{stats.totalRoutes}</p>
                  <p className="text-xs text-muted-foreground">РњР°СЂС€СЂСѓС‚РѕРІ</p>
                </div>
              </div>
            </div>

            <div className="bg-card border rounded-xl p-4">
              <div className="flex items-center gap-3">
                <div className="p-2 rounded-lg bg-orange-500/10">
                  <Truck className="h-5 w-5 text-orange-500" />
                </div>
                <div>
                  <p className="text-2xl font-bold">{stats.inProgress}</p>
                  <p className="text-xs text-muted-foreground">Р’ РїСѓС‚Рё</p>
                </div>
              </div>
            </div>

            <div className="bg-card border rounded-xl p-4">
              <div className="flex items-center gap-3">
                <div className="p-2 rounded-lg bg-green-500/10">
                  <TrendingUp className="h-5 w-5 text-green-500" />
                </div>
                <div>
                  <p className="text-2xl font-bold">{stats.totalRevenue.toLocaleString()}в‚Ѕ</p>
                  <p className="text-xs text-muted-foreground">Р’С‹СЂСѓС‡РєР°</p>
                </div>
              </div>
            </div>

            <div className="bg-card border rounded-xl p-4">
              <div className="flex items-center gap-3">
                <div className="p-2 rounded-lg bg-purple-500/10">
                  <Package className="h-5 w-5 text-purple-500" />
                </div>
                <div>
                  <p className="text-2xl font-bold">{(stats.totalCapacity / 1000).toFixed(1)}С‚</p>
                  <p className="text-xs text-muted-foreground">РЎРІРѕР±РѕРґРЅРѕ РґР»СЏ РґРѕРіСЂСѓР·Р°</p>
                </div>
              </div>
            </div>
          </div>

          <div className="flex items-center gap-4">
            <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as any)}>
              <TabsList>
                <TabsTrigger value="active" className="gap-2">
                  <Clock className="h-4 w-4" />
                  РђРєС‚РёРІРЅС‹Рµ
                </TabsTrigger>
                <TabsTrigger value="completed" className="gap-2">
                  <CheckCircle2 className="h-4 w-4" />
                  Р—Р°РІРµСЂС€С‘РЅРЅС‹Рµ
                </TabsTrigger>
              </TabsList>
            </Tabs>

            <div className="flex-1" />

            <div className="relative w-64">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
              <Input
                placeholder="РџРѕРёСЃРє..."
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="pl-9"
              />
            </div>

            <Button variant="outline" size="icon" onClick={fetchRoutes}>
              <RefreshCw className={`h-4 w-4 ${isLoading ? "animate-spin" : ""}`} />
            </Button>
          </div>

          {isLoading ? (
            <div className="flex justify-center py-20">
              <Loader2 className="h-8 w-8 animate-spin" />
            </div>
          ) : filteredRoutes.length === 0 ? (
            <div className="text-center py-20 bg-muted/30 rounded-xl border border-dashed">
              <Route className="h-12 w-12 mx-auto mb-4 text-muted-foreground/50" />
              <p className="text-lg font-medium">РќРµС‚ РјР°СЂС€СЂСѓС‚РѕРІ</p>
              <p className="text-muted-foreground">
                {activeTab === "active"
                  ? "РћС„РѕСЂРјРёС‚Рµ СЂРµР№СЃ РІ СЂР°Р·РґРµР»Рµ Р—Р°РєР°Р·С‹"
                  : "Р—Р°РІРµСЂС€С‘РЅРЅС‹Рµ РјР°СЂС€СЂСѓС‚С‹ РїРѕСЏРІСЏС‚СЃСЏ Р·РґРµСЃСЊ"}
              </p>
            </div>
          ) : (
            <div className="grid gap-4">
              {filteredRoutes.map((route) => (
                <ActiveRouteCard
                  key={route.id}
                  route={route}
                  onAddLoad={() => setAddLoadRoute(route)}
                  onRefresh={fetchRoutes}
                />
              ))}
            </div>
          )}
        </main>
      </div>

      <AddLoadDialog
        open={!!addLoadRoute}
        onOpenChange={(open) => !open && setAddLoadRoute(null)}
        route={addLoadRoute}
        onSuccess={() => {
          setAddLoadRoute(null)
          fetchRoutes()
        }}
      />
    </div>
  )
}
