// components/dashboard/map/DashboardMap.tsx

"use client"

import { useRef, useState, useCallback, useEffect } from "react"
import "leaflet/dist/leaflet.css"

// Hooks
import { useMapInstance } from "./hooks/useMapInstance"
import { useMapData } from "./hooks/useMapData"
import { useRouteAnimation } from "./hooks/useRouteAnimation"

// Layers
import { BaseMarker } from "./layers/BaseMarker"
import { WaypointMarkers } from "./layers/WaypointMarkers"
import { DriverMarkers } from "./layers/DriverMarkers"

// Panels
import { DriversPanel } from "./panels/DriversPanel"
import { StatsOverlay } from "./panels/StatsOverlay"

// Styles
import { mapStyles } from "./styles"

export default function DashboardMap() {
  const mapContainerRef = useRef<HTMLDivElement>(null)

  // РЎРѕСЃС‚РѕСЏРЅРёРµ UI
  const [showDriversList, setShowDriversList] = useState(false)
  const [showRoutes, setShowRoutes] = useState(true)
  const [selectedDriver, setSelectedDriver] = useState<string | null>(null)

  // вњ… РРЎРџР РђР’Р›Р•РќРћ: map С‚РµРїРµСЂСЊ РёР· state, Р° РЅРµ РёР· ref
  const { map, canvasRef, flyTo, fitBounds } = useMapInstance({
    containerRef: mapContainerRef,
  })

  // РҐСѓРє РґР°РЅРЅС‹С…
  const {
    drivers,
    routes,
    base,
    baseWarning,
    stats,
    totalActiveKm,
    isLoading,
    lastUpdate,
    refresh,
  } = useMapData()

  // вњ… РРЎРџР РђР’Р›Р•РќРћ: РїРµСЂРµРґР°С‘Рј map (state) РІРјРµСЃС‚Рѕ mapRef.current
  useRouteAnimation({
    canvasRef,
    map,
    routes,
    enabled: showRoutes,
  })

  // Р¦РµРЅС‚СЂРёСЂРѕРІР°РЅРёРµ РїСЂРё РїРµСЂРІРѕР№ Р·Р°РіСЂСѓР·РєРµ
  useEffect(() => {
    // вњ… РРЎРџР РђР’Р›Р•РќРћ: РїСЂРѕРІРµСЂСЏРµРј С‡С‚Рѕ РєР°СЂС‚Р° РіРѕС‚РѕРІР°
    if (!map || !lastUpdate) return

    const allPoints: [number, number][] = []
    if (base) allPoints.push(base.coordinates)
    drivers.forEach((d) => {
      if (d.latitude && d.longitude) {
        allPoints.push([d.latitude, d.longitude])
      }
    })

    if (allPoints.length > 0) {
      fitBounds(allPoints)
    }
  }, [map, base, drivers, lastUpdate, fitBounds])

  // Handlers
  const handleFlyToBase = useCallback(() => {
    if (base) flyTo(base.coordinates, 14)
  }, [base, flyTo])

  const handleFlyToDriver = useCallback(
    (lat: number, lng: number) => {
      flyTo([lat, lng], 14)
    },
    [flyTo]
  )

  const handleSelectDriver = useCallback((id: string) => {
    setSelectedDriver(id)
  }, [])

  return (
    <div className="relative h-full w-full bg-[#0a0a0a] overflow-hidden">
      {/* Р“Р»РѕР±Р°Р»СЊРЅС‹Рµ СЃС‚РёР»Рё */}
      <style jsx global>
        {mapStyles}
      </style>

      {/* РљРѕРЅС‚РµР№РЅРµСЂ РєР°СЂС‚С‹ */}
      <div ref={mapContainerRef} className="h-full w-full z-0" />

      {/* вњ… РРЎРџР РђР’Р›Р•РќРћ: РїРµСЂРµРґР°С‘Рј map (state) РІРјРµСЃС‚Рѕ mapRef.current */}
      <BaseMarker map={map} base={base} />
      <WaypointMarkers map={map} routes={routes} enabled={showRoutes} />
      <DriverMarkers
        map={map}
        drivers={drivers}
        selectedDriverId={selectedDriver}
        onSelectDriver={handleSelectDriver}
      />

      {/* РћРІРµСЂР»РµР№ СЃС‚Р°С‚РёСЃС‚РёРєРё */}
      <StatsOverlay
        base={base}
        baseWarning={baseWarning}
        stats={stats}
        routesCount={routes.length}
        totalActiveKm={totalActiveKm}
        showRoutes={showRoutes}
        onToggleRoutes={() => setShowRoutes(!showRoutes)}
        onToggleDriversPanel={() => setShowDriversList(!showDriversList)}
        showDriversPanel={showDriversList}
        isLoading={isLoading}
        lastUpdate={lastUpdate}
        onRefresh={refresh}
        onFlyToBase={handleFlyToBase}
      />

      {/* Р‘РѕРєРѕРІР°СЏ РїР°РЅРµР»СЊ РІРѕРґРёС‚РµР»РµР№ */}
      <DriversPanel
        isOpen={showDriversList}
        onClose={() => setShowDriversList(false)}
        drivers={drivers}
        stats={stats}
        selectedDriverId={selectedDriver}
        onSelectDriver={setSelectedDriver}
        onFlyToDriver={handleFlyToDriver}
      />
    </div>
  )
}
