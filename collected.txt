import { useState, useEffect, useCallback } from 'react'

export function useFleet() {
  const [drivers, setDrivers] = useState<any[]>([])
  const [vehicles, setVehicles] = useState<any[]>([])
  const [stats, setStats] = useState<any>(null)
  const [isLoading, setIsLoading] = useState(true)

  const fetchAll = useCallback(async () => {
    setIsLoading(true)
    try {
      const [driversRes, vehiclesRes, statsRes] = await Promise.all([
        fetch('/api/drivers'),
        fetch('/api/vehicles'),
        fetch('/api/fleet/stats'),
      ])
      
      const [driversData, vehiclesData, statsData] = await Promise.all([
        driversRes.json(),
        vehiclesRes.json(),
        statsRes.json(),
      ])
      
      if (driversData.success) setDrivers(driversData.drivers)
      if (vehiclesData.success) setVehicles(vehiclesData.vehicles)
      if (statsData.success) setStats(statsData)
    } catch (e) {
      console.error('useFleet error:', e)
    } finally {
      setIsLoading(false)
    }
  }, [])

  useEffect(() => {
    fetchAll()
  }, [fetchAll])

  // Р”РѕР±Р°РІРёС‚СЊ РІРѕРґРёС‚РµР»СЏ
  const addDriver = async (data: any) => {
    const res = await fetch('/api/drivers', {

      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
    return result.driver
  }

  // РЈРґР°Р»РёС‚СЊ РІРѕРґРёС‚РµР»СЏ
  const deleteDriver = async (id: string) => {
    const res = await fetch(`/api/drivers/${id}`, { method: 'DELETE' })

    if (!result.success) throw new Error(result.error)
    await fetchAll()
  }

  // Р”РѕР±Р°РІРёС‚СЊ РўРЎ
  const addVehicle = async (data: any) => {
    const res = await fetch('/api/vehicles', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
    return result.vehicle
  }

  // РЈРґР°Р»РёС‚СЊ РўРЎ
  const deleteVehicle = async (id: string) => {
    const res = await fetch(`/api/vehicles/${id}`, { method: 'DELETE' })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
  }

import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function GET(req: NextRequest) {
  const { searchParams } = new URL(req.url)
  const minCapacity = parseInt(searchParams.get('minCapacity') || '0')
  const status = searchParams.get('status')

  try {
    const where: any = {}
    
    if (status) {
      where.status = status
    }
    
    if (minCapacity > 0) {
      where.capacity = { gte: minCapacity }
    }

    const vehicles = await prisma.vehicle.findMany({  
      where,
      orderBy: [
        { status: 'asc' },
        { capacity: 'asc' },
      ],
      take: 50,
    })

    const vehiclesWithDrivers = await Promise.all(    
      vehicles.map(async (v) => {
        let driver = null
        if (v.driverId) {
          driver = await prisma.driver.findUnique({   
            where: { id: v.driverId },
            select: {
              id: true,
              name: true,
              phone: true,
              status: true,
              currentLocation: true,
            }
          })
        }
        return { ...v, driver }
      })
    )

    // РРЎРџР РђР’Р›Р•РќРР•: Р’РѕР·РІСЂР°С‰Р°РµРј РѕР±СЉРµРєС‚ success: true
    return NextResponse.json({
      success: true,
      vehicles: vehiclesWithDrivers
    })     
  } catch (error: any) {
    console.error('[Fleet Vehicles] Error:', error)   
    return NextResponse.json({ 
      success: false, 
      error: error.message,
      vehicles: [] 
    }, { status: 500 })
  }
}

export async function GET() {

    // Р’СЃРµ РІРѕРґРёС‚РµР»Рё

    const activeShifts = await prisma.driverShift.findMany({
      where: { endedAt: null },
      select: {
        driverId: true,
        status: true,
        startedAt: true,
        lastStatusChangeAt: true,
      },
    })

    const shiftMap = new Map(activeShifts.map(s => [s.driverId, s]))

    // РђРєС‚РёРІРЅС‹Рµ Р·Р°РєР°Р·С‹
    const activeOrders = await prisma.order.findMany({
      where: {
        status: { in: ["confirmed", "in_transit", "loading", "unloading"] },
        assignedDriverId: { not: null },
      },
      select: {
        assignedDriverId: true,
        routeFrom: true,
        routeTo: true,
        cargoType: true,
        price: true,
        status: true,
      },
    })

    const orderMap = new Map<string, typeof activeOrders[0]>()
    activeOrders.forEach(o => {
      if (o.assignedDriverId) {
        orderMap.set(o.assignedDriverId, o)
      }
    })

    // Р¤РѕСЂРјРёСЂСѓРµРј СЃРїРёСЃРѕРє РІРѕРґРёС‚РµР»РµР№
    const now = Date.now()
    const drivers = allDrivers.map(driver => {
      const shift = shiftMap.get(driver.id)
      const order = orderMap.get(driver.id)
      
      // РћРїСЂРµРґРµР»СЏРµРј СЃС‚Р°С‚СѓСЃ
      let status = 'offline'
      let statusDuration = 0
      
      if (shift) {
        status = shift.status || 'waiting'
        statusDuration = Math.floor((now - shift.lastStatusChangeAt.getTime()) / 1000)
      } else if (driver.status === 'maintenance') {
        status = 'maintenance'
      }

      return {
        id: driver.id,
        name: driver.name,
        phone: driver.phone,
        latitude: driver.latitude,
        longitude: driver.longitude,
        status,
        statusDuration,
        vehiclePlate: driver.vehiclePlate,
        vehicleType: driver.vehicleType,
        hasOrder: !!order,
        routeFrom: order?.routeFrom,
        routeTo: order?.routeTo,
        cargoType: order?.cargoType,
        orderPrice: order?.price,
      }
    })

    // РЎС‚Р°С‚РёСЃС‚РёРєР°
    const online = drivers.filter(d => d.status !== 'offline' && d.latitude != null).length
    const inRoute = drivers.filter(d => 
      ['driving', 'in_transit', 'loading', 'unloading'].includes(d.status)
    ).length

    // Р—Р°РєР°Р·С‹ Р·Р° СЃРµРіРѕРґРЅСЏ
    const todayStart = new Date()
    todayStart.setHours(0, 0, 0, 0)

    const [completedToday, newToday, totalOrders, activeOrdersCount] = await Promise.all([
      prisma.order.count({
        where: {
          status: "delivered",
          updatedAt: { gte: todayStart },
        },
      }),
      prisma.order.count({
        where: {
          createdAt: { gte: todayStart },
        },
      }),
      prisma.order.count(),
      prisma.order.count({
        where: {
          status: { in: ["confirmed", "in_transit", "loading", "unloading"] },
        },
      }),
    ])

    return NextResponse.json({
      success: true,
      drivers,
      stats: {
        online,
        inRoute,
        total: allDrivers.length,
        orders: {
          total: totalOrders,
          active: activeOrdersCount,
          completedToday,
          newToday,
        },
        revenue: 0,
        alerts: 0,
      },
    })
  } catch (error: any) {
    console.error("[Drivers Locations] Error:", error)
    return NextResponse.json(
      { success: false, error: error.message, drivers: [], stats: {} },
      { status: 500 }
    )
  }
