  const router = useRouter()

  // Р—Р°РіСЂСѓР·РєР° СЃРѕРѕР±С‰РµРЅРёР№
  const fetchMessages = useCallback(async () => {
    if (!driver?.id) return

    try {
      const res = await fetch(`/api/chat?driverId=${driver.id}`)
      const data = await res.json()

      if (data.success) {
        setMessages(data.messages)
      }
    } catch (error) {
      console.error("Failed to fetch messages:", error)
    } finally {
      setIsLoading(false)
    }
  }, [driver?.id])

    // РџРѕР»СѓС‡Р°РµРј РёРЅС„РѕСЂРјР°С†РёСЋ Рѕ РїСЂРёРІСЏР·Р°РЅРЅС‹С… РІРѕРґРёС‚РµР»СЏС…
    const drivers = await prisma.driver.findMany({
      where: {
        vehicleId: { not: null },
      },
      select: {
        id: true,
        name: true,
        vehicleId: true,
      },
    })

    const driverMap = new Map(

    )

    }))

    return NextResponse.json({

    })
  } catch (error) {

    const vehicle = await prisma.vehicle.findUnique({

        { success: false, error: "РњР°С€РёРЅР° РЅРµ РЅР°Р№РґРµРЅР°" },

    const driver = await prisma.driver.update({
      where: { id: driverId },
      data: {
        vehicleId: vehicleId,
        vehiclePlate: vehicle.plate,
        vehicleType: vehicle.type,
      },
    })

    return NextResponse.json({
