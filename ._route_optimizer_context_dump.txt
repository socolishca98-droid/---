// app/api/routes/route.ts

import { NextRequest, NextResponse } from "next/server"
import { prisma } from "@/lib/prisma"

const ACTIVE_ORDER_STATUSES = ["confirmed", "in_transit", "loading", "unloading"] as const

// GET вЂ” РїРѕР»СѓС‡РёС‚СЊ Р·Р°РєР°Р·С‹, СЃРіСЂСѓРїРїРёСЂРѕРІР°РЅРЅС‹Рµ РїРѕ routeId
export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url)
    const status = searchParams.get("status")
    const driverId = searchParams.get("driverId")

    const where: any = {
      routeId: { not: null },
    }

    if (status && status !== "all") {
      where.status = status
    }
    if (driverId) {
      where.assignedDriverId = driverId
    }

    const routes = await prisma.order.findMany({
      where,
      orderBy: { createdAt: "desc" },
      take: 100,
    })

    return NextResponse.json({ success: true, routes })
  } catch (error: any) {
    console.error("[Routes API] GET Error:", error)
    return NextResponse.json(
      { success: false, error: error.message || "Routes GET error" },
      { status: 500 },
    )
  }
}

// POST вЂ” СЃРѕР·РґР°С‚СЊ РјР°СЂС€СЂСѓС‚ РёР· РїРµСЃРѕС‡РЅРёС†С‹ (СЃ С‚СЂР°РЅР·Р°РєС†РёРµР№)
export async function POST(request: NextRequest) {
  try {
    const body = await request.json()
    const {
      vehicleId,
      driverId,
      orders,
      totalPrice,
      totalDistance,
      totalWeight,
    } = body as {
      vehicleId?: string
      driverId?: string
      orders?: any[]
      totalPrice?: number
      totalDistance?: number
      totalWeight?: number
    }

    if (!orders || !Array.isArray(orders) || orders.length === 0) {
      return NextResponse.json(
        { success: false, error: "РќРµС‚ Р·Р°РєР°Р·РѕРІ РґР»СЏ РјР°СЂС€СЂСѓС‚Р°" },
        { status: 400 },
      )
    }

    // РџСЂРѕРІРµСЂРєР° РІРѕРґРёС‚РµР»СЏ
    if (driverId) {
      const driver = await prisma.driver.findUnique({
        where: { id: driverId },
        select: { id: true, status: true, name: true },
      })

      if (!driver) {
        return NextResponse.json(
          { success: false, error: "Р’РѕРґРёС‚РµР»СЊ РЅРµ РЅР°Р№РґРµРЅ" },
          { status: 404 },
        )
      }

      if (driver.status === "busy") {
        return NextResponse.json(
          {
            success: false,
            error: `Р’РѕРґРёС‚РµР»СЊ ${driver.name} СѓР¶Рµ Р·Р°РЅСЏС‚ РґСЂСѓРіРёРј СЂРµР№СЃРѕРј`,
          },
          { status: 400 },
        )
      }

      if (driver.status === "maintenance") {
        return NextResponse.json(
          { success: false, error: `Р’РѕРґРёС‚РµР»СЊ ${driver.name} РЅР° РўРћ` },
          { status: 400 },
        )
      }
    }

    // РџСЂРѕРІРµСЂРєР° РјР°С€РёРЅС‹
    if (vehicleId) {
      const vehicle = await prisma.vehicle.findUnique({
        where: { id: vehicleId },
        select: { id: true, status: true, plate: true },
      })

      if (!vehicle) {
        return NextResponse.json(
          { success: false, error: "РњР°С€РёРЅР° РЅРµ РЅР°Р№РґРµРЅР°" },
          { status: 404 },
        )
      }

      if (vehicle.status === "in_use") {
        return NextResponse.json(
          {
            success: false,
            error: `РњР°С€РёРЅР° ${vehicle.plate} СѓР¶Рµ РІ СЂРµР№СЃРµ`,
          },
          { status: 400 },
        )
      }

      if (vehicle.status === "maintenance") {
        return NextResponse.json(
          { success: false, error: `РњР°С€РёРЅР° ${vehicle.plate} РЅР° РўРћ` },
          { status: 400 },
        )
      }
    }

    const routeId = `route-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`

    const createdOrders = await prisma.$transaction(async (tx) => {
      const created: any[] = []

      for (const orderData of orders) {
        const order = await tx.order.create({
          data: {
            source: orderData.source || "ATI",
            sourceId: orderData.atiLoadId || orderData.atiCacheId || orderData.id || null,

            routeFrom: orderData.routeFrom || orderData.from,
            routeTo: orderData.routeTo || orderData.to,
            distance: orderData.distance || 0,
            weight: orderData.weight || 0,
            volume: orderData.volume || null,

            cargoType: orderData.cargoType || orderData.cargo || "Р“СЂСѓР·",
            price: orderData.price || null,

            clientName:
              orderData.clientName || orderData.clientCompany || orderData.company || null,
            clientContact: orderData.clientContact || orderData.clientPhone || orderData.phone || "",

            status: "confirmed",
            assignedDriverId: driverId || null,
            assignedVehicleId: vehicleId || null,
            routeId,
            deadline:
              orderData.deadline != null
                ? new Date(orderData.deadline)
                : new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
          },
        })

        created.push(order)

        const cacheId = orderData.atiCacheId || orderData.id
        if (cacheId) {
          await tx.atiCache
            .update({
              where: { id: cacheId },
              data: { status: "imported" },
            })
            .catch(() => {})
        }
      }

      if (driverId) {
        await tx.driver.update({
          where: { id: driverId },
          data: { status: "busy" },
        })
      }

      if (vehicleId) {
        await tx.vehicle.update({
          where: { id: vehicleId },
          data: { status: "in_use" },
        })
      }

      return created
    })

    return NextResponse.json({
      success: true,
      routeId,
      ordersCount: createdOrders.length,
      orderIds: createdOrders.map((o) => o.id),
      totalPrice,
      totalDistance,
      totalWeight,
    })
  } catch (error: any) {
    console.error("[Routes API] POST Error:", error)
    return NextResponse.json(
      { success: false, error: error.message || "РћС€РёР±РєР° СЃРѕР·РґР°РЅРёСЏ РјР°СЂС€СЂСѓС‚Р°" },
      { status: 500 },
    )
  }
}
