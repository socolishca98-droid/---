      'processing',
      'loading',
      'unloading',
    ]

    const historyStatuses = ['delivered', 'cancelled']

    const orders = await prisma.order.findMany({
      where: {
        assignedDriverId: driverId,
        status: {
          in: status === 'history' ? historyStatuses : activeStatuses,

      orderBy: {

      },
      _count: true,
      _sum: {
        price: true,
        distance: true,
      },
    })

    return NextResponse.json({
      success: true,
      orders,
      stats: {
        completedOrders: stats._count || 0,
        totalEarnings: stats._sum.price || 0,

const DRIVING_LIMIT_SECONDS = 4.5 * 3600

export async function GET(req: NextRequest) {
  try {
    const driverId = req.nextUrl.searchParams.get("driverId")

    if (!driverId) {
      return NextResponse.json(
        { success: false, error: "driverId required" },
        { status: 400 }
      )
    }

    const shift = await prisma.driverShift.findFirst({
      where: { driverId, endedAt: null },
      orderBy: { startedAt: "desc" },
    })

    if (!shift) {
      return NextResponse.json({ success: true, shift: null })
    }

    const now = new Date()
    const statusDuration = Math.floor(
      (now.getTime() - shift.lastStatusChangeAt.getTime()) / 1000
    )

    let currentDrivingSeconds = shift.drivingSinceRestSeconds
    let currentRestingSeconds = shift.totalRestingSeconds
    let currentLoadingSeconds = shift.totalLoadingSeconds
    let currentWaitingSeconds = shift.totalWaitingSeconds

    switch (shift.status) {
      case "driving":
        currentDrivingSeconds += statusDuration
        break
      case "resting":
      case "sleeping":
        currentRestingSeconds += statusDuration
        break
      case "loading":
      case "unloading":
        currentLoadingSeconds += statusDuration
        break
      case "waiting":
      case "fueling":
        currentWaitingSeconds += statusDuration
        break
    }

    const totalDrivingSeconds = shift.totalDrivingSeconds
    const totalShiftSeconds =
      totalDrivingSeconds +
      currentRestingSeconds +
      currentLoadingSeconds +
      currentWaitingSeconds

    const drivingProgress = Math.min(
      100,
      (currentDrivingSeconds / DRIVING_LIMIT_SECONDS) * 100
    )

    return NextResponse.json({
      success: true,
      shift: {
        id: shift.id,
        status: shift.status,
        startedAt: shift.startedAt,
        statusDuration,
        currentDrivingSeconds,
        currentRestingSeconds,
        currentLoadingSeconds,
        currentWaitingSeconds,
        totalShiftSeconds,
        drivingProgress,
        drivingRemaining: Math.max(
          0,
          DRIVING_LIMIT_SECONDS - currentDrivingSeconds
        ),
        needsRest: currentDrivingSeconds >= DRIVING_LIMIT_SECONDS,
        restWarning: currentDrivingSeconds >= DRIVING_LIMIT_SECONDS * 0.85,
      },
    })
  } catch (error: unknown) {
    const message = error instanceof Error ? error.message : "Unknown error"
    console.error("[Shift API] GET Error:", message)
    return NextResponse.json(
      { success: false, error: message },
      { status: 500 }
    )
  }
}

      })

        message: "Shift ended",
        summary: {
          totalDriving: endedShift.totalDrivingSeconds,
          totalResting: endedShift.totalRestingSeconds,
          totalLoading: endedShift.totalLoadingSeconds,
          totalWaiting: endedShift.totalWaitingSeconds,
        },
      })

import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url)
    const driverId = searchParams.get('driverId')

    if (!driverId) {
      return NextResponse.json(
        { success: false, error: 'driverId РѕР±СЏР·Р°С‚РµР»РµРЅ' },
        { status: 400 }
      )
    }

    const driver = await prisma.driver.findUnique({
      where: { id: driverId },
      select: {
        id: true,
        name: true,
        phone: true,
        vehicleType: true,
        vehiclePlate: true,
        status: true,
        latitude: true,
        longitude: true,
        lastGpsUpdate: true,
        ordersCompleted: true,
        rating: true,
        licenseNumber: true,
        licenseExpiry: true,
        medicalExpiry: true,
        hiredAt: true,
        createdAt: true,
      }
    })

    if (!driver) {
      return NextResponse.json(
        { success: false, error: 'Р’РѕРґРёС‚РµР»СЊ РЅРµ РЅР°Р№РґРµРЅ' },
        { status: 404 }
      )
    }

    // РџРѕР»СѓС‡Р°РµРј Р°РєС‚РёРІРЅСѓСЋ СЃРјРµРЅСѓ
    const activeShift = await prisma.driverShift.findFirst({
      where: {
        driverId: driverId,
        endedAt: null
      },
      orderBy: {
        startedAt: 'desc'
      }
    })

    // РЎС‚Р°С‚РёСЃС‚РёРєР° Р·Р° СЃРµРіРѕРґРЅСЏ
    const today = new Date()
    today.setHours(0, 0, 0, 0)

    const todayStats = await prisma.order.aggregate({
      where: {
        assignedDriverId: driverId,
        status: 'delivered',
        updatedAt: { gte: today }
      },
      _count: true,
      _sum: {
        distance: true,
        price: true
      }
    })

    return NextResponse.json({
      success: true,
      driver,
      activeShift: activeShift || null,
      todayStats: {
        ordersDelivered: todayStats._count || 0,
        distanceKm: todayStats._sum.distance || 0,
        earnings: todayStats._sum.price || 0
      }
    })

  } catch (error: unknown) {
    const message = error instanceof Error ? error.message : 'Unknown error'
    console.error('[Me API] Error:', message)
    return NextResponse.json(
      { success: false, error: message },
      { status: 500 }
    )
  }
}

import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select"

interface AssignDriverDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  onAssign: (driverId: string) => Promise<void>
}

export function AssignDriverDialog({ open, onOpenChange, onAssign }: AssignDriverDialogProps) {
  const [drivers, setDrivers] = useState<any[]>([])
  const [selectedDriver, setSelectedDriver] = useState("")
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    if (open) {
      fetch('/api/drivers?status=available')
        .then(res => res.json())
        .then(data => {
          if (data.success) setDrivers(data.drivers)
        })
    }
  }, [open])

  const handleAssign = async () => {
    if (!selectedDriver) return
    setLoading(true)
    try {
      await onAssign(selectedDriver)
      onOpenChange(false)
    } finally {
      setLoading(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader>

// hooks/use-fleet.ts

import { useState, useEffect, useCallback } from 'react'

export function useFleet() {
  const [drivers, setDrivers] = useState<any[]>([])
  const [vehicles, setVehicles] = useState<any[]>([])
  const [stats, setStats] = useState<any>(null)
  const [isLoading, setIsLoading] = useState(true)

  const fetchAll = useCallback(async () => {
    setIsLoading(true)
    try {
      const [driversRes, vehiclesRes, statsRes] = await Promise.all([
        fetch('/api/drivers'),
        fetch('/api/vehicles'),
        fetch('/api/fleet/stats'),
      ])
      
      const [driversData, vehiclesData, statsData] = await Promise.all([
        driversRes.json(),
        vehiclesRes.json(),
        statsRes.json(),
      ])
      
      if (driversData.success) setDrivers(driversData.drivers)
      if (vehiclesData.success) setVehicles(vehiclesData.vehicles)
      if (statsData.success) setStats(statsData)
    } catch (e) {
      console.error('useFleet error:', e)
    } finally {
      setIsLoading(false)
    }
  }, [])

  useEffect(() => {
    fetchAll()
  }, [fetchAll])

  // Р”РѕР±Р°РІРёС‚СЊ РІРѕРґРёС‚РµР»СЏ
  const addDriver = async (data: any) => {
    const res = await fetch('/api/drivers', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
    return result.driver
  }

  // РЈРґР°Р»РёС‚СЊ РІРѕРґРёС‚РµР»СЏ
  const deleteDriver = async (id: string) => {
    const res = await fetch(`/api/drivers/${id}`, { method: 'DELETE' })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
  }

  // Р”РѕР±Р°РІРёС‚СЊ РўРЎ
  const addVehicle = async (data: any) => {
    const res = await fetch('/api/vehicles', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
    return result.vehicle
  }

  // РЈРґР°Р»РёС‚СЊ РўРЎ
  const deleteVehicle = async (id: string) => {
    const res = await fetch(`/api/vehicles/${id}`, { method: 'DELETE' })
    const result = await res.json()
    if (!result.success) throw new Error(result.error)
    await fetchAll()
  }

  // РЎРІРѕР±РѕРґРЅС‹Рµ РўРЎ (РґР»СЏ РІС‹РїР°РґР°С€РєРё)
  const getAvailableVehicles = async () => {
    const res = await fetch('/api/vehicles?available=true')
    const data = await res.json()
    return data.success ? data.vehicles : []
  }

  return {
    drivers,
    vehicles,
    stats,
    isLoading,
    refresh: fetchAll,
    addDriver,
    deleteDriver,
    addVehicle,
    deleteVehicle,
    getAvailableVehicles,
  }
}
